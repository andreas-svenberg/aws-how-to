# rejected-traffic-query.txt
# Logs Insights query for the flow log in /<prefix>/vpc/flowlogs.
# CloudWatch, Logs, Log Analytics: select the log group, paste the query, run.
# Shows the 20 latest rejected connections. Field names are expected to be
# discovered automatically for flow logs in the default format.

fields @timestamp, srcAddr, dstAddr, dstPort, action
| filter action = 'REJECT'
| sort @timestamp desc
| limit 20
